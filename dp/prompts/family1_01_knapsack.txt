LARGE code block:
  treasures = [
      ("Gold", 4, 10), ("Ruby", 3, 7), ("Diamond", 5, 12),
      ("Emerald", 1, 3), ("Platinum", 2, 5)
  ]

  def best_haul(treasures, bag_limit, count):
      # BASE CASE: no items left OR no capacity
      if count == 0 or bag_limit == 0:
          return 0

      # Unpack the current item
      name, weight, value = treasures[count - 1]

      # Does this item fit?
      if weight <= bag_limit:
          # CHOICE! Grab or skip, take the better one
          grab = value + best_haul(treasures, bag_limit - weight, count - 1)
          skip = best_haul(treasures, bag_limit, count - 1)
          return max(grab, skip)
      else:
          # Doesn't fit. Skip only.
          return best_haul(treasures, bag_limit, count - 1)

  "skip = ..." ← RED path from choice diagram
  "return max(grab, skip)" ← "Take the better outcome"
  "else: return skip" ← "Doesn't fit, forced to skip"

  def best_haul(treasures, bag_limit, count, memo):
      if (count, bag_limit) in memo:           # ALREADY SOLVED? Return it!

      # ... same recursive code as before ...
      memo[(count, bag_limit)] = result         # STORE the result
      return result

Label: "Memoization turns exponential O(2^count) into polynomial O(count * bag_limit). Every unique (count, bag_limit) pair solved at most once."

Show 3 code blocks side by side (or stacked), with DIFF HIGHLIGHTS showing what changes:

VERSION 1, RECURSION (label: "correct but SLOW, O(2^count)"):

      if count == 0 or bag_limit == 0:
          return 0
      name, weight, value = treasures[count - 1]
      if weight <= bag_limit:
          grab = value + best_haul(treasures, bag_limit - weight, count - 1)
          skip = best_haul(treasures, bag_limit, count - 1)

          return best_haul(treasures, bag_limit, count - 1)

VERSION 2, MEMOIZATION (label: "add 3 lines, O(count * bag_limit)"):
  Highlight the 3 NEW lines in GREEN:
  def best_haul(treasures, bag_limit, count, memo={}):
      if (count, bag_limit) in memo: return memo[(count, bag_limit)]   # CHECK cache
      if count == 0 or bag_limit == 0:
          return 0
      name, weight, value = treasures[count - 1]
      if weight <= bag_limit:
          grab = value + best_haul(treasures, bag_limit - weight, count - 1, memo)
          skip = best_haul(treasures, bag_limit, count - 1, memo)
          result = max(grab, skip)
      else:
          result = best_haul(treasures, bag_limit, count - 1, memo)
      memo[(count, bag_limit)] = result                                # STORE result
      return result

VERSION 3, TABULATION (label: "bottom-up, O(count * bag_limit), no recursion"):
  def best_haul(treasures, bag_limit):
      table = [[0] * (bag_limit + 1) for _ in range(len(treasures) + 1)]
      for row in range(1, len(treasures) + 1):
          name, weight, value = treasures[row - 1]
          without_this = table[row - 1]
          for cap in range(1, bag_limit + 1):
              if weight <= cap:
                  table[row][cap] = max(value + without_this[cap - weight],
                                        without_this[cap])
              else:
                  table[row][cap] = without_this[cap]
      return table[len(treasures)][bag_limit]
