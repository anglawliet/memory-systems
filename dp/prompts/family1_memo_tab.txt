  dp[3] = max(5, dp[0]+4) = max(5, 4) = 5

Highlight: "RIGHT TO LEFT prevents reuse."
"dp[cap-weight] still has LAST round's value, not this round's."
"That's what makes it 0/1: each item used at most once."

SECTION 2, "THE CODE" (gray header, about 45%):

  def max_value(treasures, bag_limit):
      dp = [0] * (bag_limit + 1)

      for name, weight, value in treasures:
          for cap in range(bag_limit, weight - 1, -1):
              dp[cap] = max(dp[cap], dp[cap - weight] + value)

      return dp[bag_limit]

Annotations:
  dp = [0] * (bag_limit + 1)                <- "All zeros: no items yet."
  for name, weight, value in treasures       <- "One item at a time."
  range(bag_limit, weight-1, -1)             <- "RIGHT TO LEFT! No reuse."
  dp[cap - weight] + value                   <- "Grab: value + best for remainder."
  max(dp[cap], ...)                          <- "Grab or skip, keep the better."

Gold comparison box:
  "0/1 Knapsack: RIGHT TO LEFT (no reuse)"
  "Unbounded:    LEFT TO RIGHT (allows reuse)"
  "One loop direction is the entire difference."

  def can_reach(nums, target, index, cache):
      if target == 0:
          return True
      if index == len(nums) or target < 0:
          return False
      if (index, target) in cache:
          return cache[(index, target)]

      if nums[index] > target:
          result = can_reach(nums, target, index + 1, cache)
      else:
          take = can_reach(nums, target - nums[index], index + 1, cache)
          leave = can_reach(nums, target, index + 1, cache)
          result = take or leave

      cache[(index, target)] = result
      return result

"To minimize: find S closest to total/2 = 11.5."

"This is SUBSET SUM in disguise!"
"Find all achievable sums up to total/2. Pick the closest."

Show achievable sums:
  S=0 (pick nothing), S=1, S=5, S=6, S=7, S=11, S=12
  Closest to 11.5 = S=11.
  Difference = |23 - 2*11| = 1.

SECTION 2, "THE CODE" (gray header, about 55%):

  def min_difference(nums):
      total = sum(nums)
      half = total // 2
      cache = {}

      def can_reach(target, index):
          if target == 0:
              return True
          if index == len(nums) or target < 0:
              return False
          if (index, target) in cache:
              return cache[(index, target)]
          result = (can_reach(target - nums[index], index + 1) or
                    can_reach(target, index + 1))
          cache[(index, target)] = result
          return result

      for s in range(half, -1, -1):
          if can_reach(s, 0):
              return total - 2 * s

Kit: "Find achievable sums using subset sum with memoization.
Pick the one closest to half. That minimizes the difference."

  def min_difference(nums):
      total = sum(nums)
      half = total // 2
      dp = [False] * (half + 1)
      dp[0] = True

      for num in nums:
          for t in range(half, num - 1, -1):
              if dp[t - num]:
                  dp[t] = True

      for s in range(half, -1, -1):
          if dp[s]:
              return total - 2 * s

Annotations:
  half = total // 2        <- "Only check up to half. Symmetry."
  dp[0] = True             <- "Sum of 0 is always achievable."
  RIGHT TO LEFT             <- "Each number used once."
  range(half, -1, -1)      <- "Find closest True to half."
  total - 2 * s            <- "The minimum difference."
