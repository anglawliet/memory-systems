  def lcs_length(text1, text2, i, j, cache):
      if i == len(text1) or j == len(text2):
          return 0
      if (i, j) in cache:
          return cache[(i, j)]
      if text1[i] == text2[j]:
          result = 1 + lcs_length(text1, text2, i + 1, j + 1, cache)
      else:
          skip_first = lcs_length(text1, text2, i + 1, j, cache)
          skip_second = lcs_length(text1, text2, i, j + 1, cache)
          result = max(skip_first, skip_second)
      cache[(i, j)] = result
      return result

  3. "Store result"

Two strings = two indices = 2D cache."

Gold box: "Time: O(m * n) instead of O(2^(m+n)).
One entry per unique (i, j) pair."

  def lcs_length(text1, text2):
      rows = len(text1)
      cols = len(text2)
      table = [[0] * (cols + 1) for _ in range(rows + 1)]
      for i in range(1, rows + 1):
          for j in range(1, cols + 1):
              if text1[i - 1] == text2[j - 1]:
                  table[i][j] = table[i - 1][j - 1] + 1

  def count_paths(rows, cols, r, c, cache):
      if r == rows - 1 and c == cols - 1:
          return 1
      if r >= rows or c >= cols:
          return 0
      if (r, c) in cache:
          return cache[(r, c)]
      go_down = count_paths(rows, cols, r + 1, c, cache)
      go_right = count_paths(rows, cols, r, c + 1, cache)
      cache[(r, c)] = go_down + go_right
      return cache[(r, c)]

Kit: "No match/no-match here. Always add down + right.
Simplest 2D DP. Pure addition."

      table = [[1] * cols for _ in range(rows)]
      for r in range(1, rows):
          for c in range(1, cols):
              table[r][c] = table[r - 1][c] + table[r][c - 1]

  def min_edits(word1, word2, i, j, cache):
      if i == len(word1):
          return len(word2) - j
      if j == len(word2):
          return len(word1) - i
      if (i, j) in cache:
          return cache[(i, j)]
      if word1[i] == word2[j]:
          result = min_edits(word1, word2, i + 1, j + 1, cache)
      else:
          replace = 1 + min_edits(word1, word2, i + 1, j + 1, cache)
          delete = 1 + min_edits(word1, word2, i + 1, j, cache)
          insert = 1 + min_edits(word1, word2, i, j + 1, cache)
          result = min(replace, delete, insert)
      cache[(i, j)] = result
      return result

Kit: "Match = free, advance both. No match = 1 + min(replace, delete, insert)."

Gold box: "Base cases: ran out of word1? Insert remaining.
Ran out of word2? Delete remaining."

      rows = len(word1)
      cols = len(word2)
      table = [[0] * (cols + 1) for _ in range(rows + 1)]
      for i in range(rows + 1):
          table[i][0] = i
      for j in range(cols + 1):
          table[0][j] = j
      for i in range(1, rows + 1):

SECTION 2, "THE CODE" (gray header, about 55%):

  def can_interleave(s1, s2, s3, i, j, cache):
      if i + j == len(s3):
          return i == len(s1) and j == len(s2)
      if (i, j) in cache:
          return cache[(i, j)]
      result = False
      if i < len(s1) and s1[i] == s3[i + j]:
          result = can_interleave(s1, s2, s3, i + 1, j, cache)
      if not result and j < len(s2) and s2[j] == s3[i + j]:
          result = can_interleave(s1, s2, s3, i, j + 1, cache)
      cache[(i, j)] = result
      return result

Kit: "Boolean DP. True or False, not a number.
Can we weave s1 and s2 into s3 without reordering?"

Gold box: "The twist: target char = s3[i + j].
i tracks s1, j tracks s2, but the target auto-advances."

             OR (from_s2: table[i][j-1] AND s2[j-1]==s3[i+j-1])

Label each: "s1 match?" / "s2 match?"

SECTION 2, "THE CODE" (gray header, about 50%):

  def can_interleave(s1, s2, s3):
      if len(s1) + len(s2) != len(s3):
          return False
      rows = len(s1)
      cols = len(s2)
      table = [[False] * (cols + 1) for _ in range(rows + 1)]
      table[0][0] = True
      for j in range(1, cols + 1):
          table[0][j] = table[0][j - 1] and s2[j - 1] == s3[j - 1]
      for i in range(1, rows + 1):
          table[i][0] = table[i - 1][0] and s1[i - 1] == s3[i - 1]
      for i in range(1, rows + 1):
          for j in range(1, cols + 1):
              from_s1 = table[i - 1][j] and s1[i - 1] == s3[i + j - 1]
              from_s2 = table[i][j - 1] and s2[j - 1] == s3[i + j - 1]
              table[i][j] = from_s1 or from_s2
      return table[rows][cols]

Gold box: "Vs LCS: LCS looks for matches and counts.
Interleaving checks if a specific target can be built. Boolean, not numeric."

SECTION 2, "THE CODE" (gray header, about 55%):

  def count_subsequences(s, t, i, j, cache):
      if j == len(t):
          return 1
      if i == len(s):
          return 0
      if (i, j) in cache:
          return cache[(i, j)]
      skip_char = count_subsequences(s, t, i + 1, j, cache)
      if s[i] == t[j]:
          use_char = count_subsequences(s, t, i + 1, j + 1, cache)
          result = use_char + skip_char
      else:
          result = skip_char
      cache[(i, j)] = result
      return result

Kit: "Match = use + skip. No match = skip only.
Like 0/1 knapsack, but for characters."

Gold box: "Base cases:
  j reaches end of t? Found a complete match! Return 1.
  i reaches end of s? Ran out of characters. Return 0."

SECTION 2, "THE CODE" (gray header, about 50%):

  def count_subsequences(s, t):
      rows = len(s)
      cols = len(t)
      table = [[0] * (cols + 1) for _ in range(rows + 1)]
      for i in range(rows + 1):
          table[i][0] = 1
      for i in range(1, rows + 1):
          for j in range(1, cols + 1):
              table[i][j] = table[i - 1][j]
              if s[i - 1] == t[j - 1]:
                  table[i][j] += table[i - 1][j - 1]
      return table[rows][cols]

Gold box: "Vs LCS: LCS picks max(up, left) on mismatch.
Distinct Subs always takes up, and ADDS diagonal on match.
It is counting, not maximizing."
