   1  from collections import deque
   2
   3  def topo_sort(n, edges):
   4      graph = [[] for _ in range(n)]
   5      in_deg = [0] * n
   6      for u, v in edges:
   7          graph[u].append(v)
   8          in_deg[v] += 1
   9
  10      q = deque(i for i in range(n) if in_deg[i] == 0)
