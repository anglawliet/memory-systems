   1  parent = list(range(n))
   2  rank = [0] * n
   3
   4  def find(x):
   5      if parent[x] != x:
   6          parent[x] = find(parent[x])
   7      return parent[x]
   8
   9  def union(a, b):
  10      pa, pb = find(a), find(b)
  11      if pa == pb:
  12          return False
  13      if rank[pa] < rank[pb]:
  14          pa, pb = pb, pa
  15      parent[pb] = pa
  16      if rank[pa] == rank[pb]:
  17          rank[pa] += 1
  18      return True
